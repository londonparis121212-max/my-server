from flask import Flask, request, jsonify
from datetime import datetime
import os

app = Flask(__name__)

# Константа — 2100 год (никогда не истекает)
EXPIRE_TIME = int(datetime(2100, 1, 1).timestamp() * 1000)

@app.route('/api/activate-key', methods=['POST', 'GET'])
def activate_key():
    """Основной эндпоинт для активации ключа"""
    if request.method == 'POST':
        # Логируем запрос для отладки (в Railway логи будут видны)
        data = request.get_json()
        key = data.get('key', '') if data else ''
        device_id = data.get('deviceId', '') if data else ''
        print(f"📥 POST запрос: key={key}, deviceId={device_id}")
    
    # Всегда возвращаем успешный ответ
    return jsonify({
        "status": "ok",
        "valid": True,
        "expiresAt": EXPIRE_TIME
    })

@app.route('/health')
def health():
    """Эндпоинт для проверки работоспособности"""
    return jsonify({"status": "healthy"})

@app.route('/')
def index():
    """Корневой эндпоинт"""
    return jsonify({
        "status": "ok",
        "valid": True
    })

# Для локального запуска
if __name__ == '__main__':
    port = int(os.environ.get('PORT', 5000))
    app.run(host='0.0.0.0', port=port, debug=False)
